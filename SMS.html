<script>
function switchTab(tab) {
['dashboard', 'allocate', 'menu', 'numbers', 'sms'].forEach(t => {
document.getElementById('view-' + t).classList.add('hidden');
document.getElementById('nav-' + t).classList.remove('text-teal-600');
});
document.getElementById('view-' + tab).classList.remove('hidden');
document.getElementById('nav-' + tab).classList.add('text-teal-600');
}

async function fetchSMS() {
const container = document.getElementById('sms-history-list');
container.innerHTML = '<p class="text-xs text-gray-500 text-center py-4">Loading messages...</p>';
try {
const res = await fetch('/api/lamix?action=messages');
const data = await res.json();
if(data.records && data.records.length > 0) {
document.getElementById('total-sms-count').innerText = data.records.length;
container.className = "space-y-2";
container.innerHTML = data.records.map(item => `
<div class="bg-white p-3 rounded-xl shadow-sm border border-gray-100 space-y-1 text-left">
<div class="flex justify-between text-xs text-gray-500">
<span><b>NUMBER:</b> ${item.number}</span>
<span class="text-teal-600 font-bold">$${item.payout}</span>
</div>
<p class="text-xs text-gray-800"><b>MESSAGE:</b> ${item.content}</p>
<span class="text-[10px] bg-teal-50 text-teal-600 px-2 py-0.5 rounded inline-block">${item.range}</span>
</div>
`).join('');
} else {
container.className = "bg-white p-8 rounded-2xl shadow-sm border border-gray-100 text-center space-y-3";
container.innerHTML = `
<div class="w-16 h-16 bg-blue-50 text-blue-400 rounded-full flex items-center justify-center mx-auto text-2xl">
<i class="fas fa-envelope-open-text"></i>
</div>
<h4 class="font-bold text-gray-800 text-sm">No SMS found</h4>
<p class="text-xs text-gray-400">Try changing the filters, or wait for new messages</p>
`;
}
} catch (err) {
container.className = "bg-white p-8 rounded-2xl shadow-sm border border-gray-100 text-center space-y-3";
container.innerHTML = `
<div class="w-16 h-16 bg-blue-50 text-blue-400 rounded-full flex items-center justify-center mx-auto text-2xl">
<i class="fas fa-envelope-open-text"></i>
</div>
<h4 class="font-bold text-gray-800 text-sm">No SMS found</h4>
<p class="text-xs text-gray-400">Try changing the filters, or wait for new messages</p>
`;
}
}

async function fetchNumbers() {
const container = document.getElementById('numbers-list');
container.innerHTML = '<p class="text-xs text-gray-500 text-center py-2">Loading numbers...</p>';
try {
const res = await fetch('/api/lamix?action=numbers');
const data = await res.json();
if(data.records && data.records.length > 0) {
container.innerHTML = data.records.map(n => `
<div class="bg-white p-3 rounded-xl shadow-sm border flex justify-between items-center">
<div>
<h4 class="font-bold text-xs text-gray-800">${n.range}</h4>
<p class="text-[10px] text-gray-500">${n.number}</p>
</div>
<span class="text-xs text-teal-600 font-bold">$${n.payoutRate}</span>
</div>
`).join('');
}
} catch (err) {
container.innerHTML = '<p class="text-xs text-red-500 text-center py-2">Failed to load numbers.</p>';
}
}
